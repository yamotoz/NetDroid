sudo apt update -y && sudo apt install -y python3 python3-pip python3-venv aircrack-ng mdk4 hcxdumptool hcxtools hashcat tcpdump arp-scan iw wireless-tools nmap iperf3 iptables net-tools curl wget git && pip install --break-system-packages -r requirements.txt && mkdir -p WordList && (test -f /usr/share/wordlists/rockyou.txt || sudo gunzip -k /usr/share/wordlists/rockyou.txt.gz 2>/dev/null) && cp /usr/share/wordlists/rockyou.txt ./WordList/ 2>/dev/null; cp /usr/share/wordlists/fasttrack.txt ./WordList/ 2>/dev/null; echo "✓ NetDroid pronto. Rode: sudo python NetDroid.py --root --auto --god --live"
# ─────────────────────────────────────────────────────────────────────
#
# O que esse comando faz, em ordem:
#   1) sudo apt update -y                      → atualiza repositório
#   2) sudo apt install -y <pacotes>           → todos os binários nativos
#   3) pip install --break-system-packages -r requirements.txt
#                                              → todas as deps Python
#   4) mkdir -p WordList                       → garante a pasta de wordlists
#   5) gunzip rockyou.txt.gz                   → descompacta se ainda gz
#   6) cp rockyou.txt fasttrack.txt → WordList → wordlists prontas para crack
#
